-- CHAIXI BAMEEKIAO — OPERATIONS WORKFLOW V1.0
-- Daily / Month-End / Year-End checklist state
-- Safe patch: does not modify Sales / Stock / PO / Production data.
-- ============================================================

begin;

create table if not exists public.backoffice_operation_checks (
    id uuid primary key default gen_random_uuid(),
    branch_id uuid not null references public.branches(id) on delete cascade,
    period_type text not null,
    period_key text not null,
    task_key text not null,
    completed_at timestamptz not null default now(),
    completed_by uuid references public.profiles(id) on delete set null,
    note text,
    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now(),
    constraint backoffice_operation_checks_period_type_check
      check (period_type in ('daily','month','year')),
    constraint backoffice_operation_checks_unique
      unique(branch_id,period_type,period_key,task_key)
);

create index if not exists idx_bo_operation_checks_branch_period
on public.backoffice_operation_checks(branch_id,period_type,period_key);

alter table public.backoffice_operation_checks enable row level security;

-- No direct table grants are required. Access is only through the RPCs below.
revoke all on table public.backoffice_operation_checks from anon, authenticated;

create or replace function public.backoffice_operations_list_v1(
    p_period_type text,
    p_period_key text
)
returns table(
    task_key text,
    completed_at timestamptz,
    completed_by uuid,
    completed_by_name text,
    note text
)
language plpgsql
security definer
set search_path=public
as $$
declare
    v_branch uuid;
begin
    select x.branch_id into v_branch from public._bo_ctx() x;
    if v_branch is null then raise exception 'BACKOFFICE_PERMISSION_DENIED'; end if;
    if p_period_type not in ('daily','month','year') then raise exception 'INVALID_PERIOD_TYPE'; end if;
    if nullif(trim(coalesce(p_period_key,'')),'') is null then raise exception 'INVALID_PERIOD_KEY'; end if;

    return query
    select c.task_key,c.completed_at,c.completed_by,
           coalesce(p.full_name,p.email,'-')::text as completed_by_name,
           c.note
    from public.backoffice_operation_checks c
    left join public.profiles p on p.id=c.completed_by
    where c.branch_id=v_branch
      and c.period_type=p_period_type
      and c.period_key=p_period_key
    order by c.completed_at;
end
$$;

create or replace function public.backoffice_operations_mark_v1(
    p_period_type text,
    p_period_key text,
    p_task_key text,
    p_note text default null
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$
declare
    v_branch uuid;
    v_user uuid;
    v_role text;
    v_row public.backoffice_operation_checks%rowtype;
begin
    select x.branch_id,x.user_id,x.role into v_branch,v_user,v_role from public._bo_ctx() x;
    if v_branch is null or v_user is null then raise exception 'BACKOFFICE_PERMISSION_DENIED'; end if;
    if lower(coalesce(v_role,'')) not in ('admin','manager') then raise exception 'BACKOFFICE_PERMISSION_DENIED'; end if;
    if p_period_type not in ('daily','month','year') then raise exception 'INVALID_PERIOD_TYPE'; end if;
    if nullif(trim(coalesce(p_period_key,'')),'') is null then raise exception 'INVALID_PERIOD_KEY'; end if;
    if nullif(trim(coalesce(p_task_key,'')),'') is null then raise exception 'INVALID_TASK_KEY'; end if;

    insert into public.backoffice_operation_checks(
        branch_id,period_type,period_key,task_key,completed_at,completed_by,note,updated_at
    ) values(
        v_branch,p_period_type,p_period_key,p_task_key,now(),v_user,
        nullif(trim(coalesce(p_note,'')),''),now()
    )
    on conflict(branch_id,period_type,period_key,task_key)
    do update set completed_at=excluded.completed_at,
                  completed_by=excluded.completed_by,
                  note=excluded.note,
                  updated_at=now()
    returning * into v_row;

    return jsonb_build_object(
        'task_key',v_row.task_key,
        'completed_at',v_row.completed_at,
        'completed_by',v_row.completed_by,
        'note',v_row.note
    );
end
$$;

create or replace function public.backoffice_operations_unmark_v1(
    p_period_type text,
    p_period_key text,
    p_task_key text
)
returns boolean
language plpgsql
security definer
set search_path=public
as $$
declare
    v_branch uuid;
    v_role text;
begin
    select x.branch_id,x.role into v_branch,v_role from public._bo_ctx() x;
    if v_branch is null then raise exception 'BACKOFFICE_PERMISSION_DENIED'; end if;
    if lower(coalesce(v_role,'')) not in ('admin','manager') then raise exception 'BACKOFFICE_PERMISSION_DENIED'; end if;

    delete from public.backoffice_operation_checks
    where branch_id=v_branch and period_type=p_period_type
      and period_key=p_period_key and task_key=p_task_key;
    return found;
end
$$;

revoke all on function public.backoffice_operations_list_v1(text,text) from public;
revoke all on function public.backoffice_operations_mark_v1(text,text,text,text) from public;
revoke all on function public.backoffice_operations_unmark_v1(text,text,text) from public;
grant execute on function public.backoffice_operations_list_v1(text,text) to authenticated;
grant execute on function public.backoffice_operations_mark_v1(text,text,text,text) to authenticated;
grant execute on function public.backoffice_operations_unmark_v1(text,text,text) to authenticated;

commit;

-- SQL Editor-safe verify (does not call _bo_ctx())
select proname as function_name
from pg_proc p join pg_namespace n on n.oid=p.pronamespace
where n.nspname='public' and proname in (
 'backoffice_operations_list_v1',
 'backoffice_operations_mark_v1',
 'backoffice_operations_unmark_v1'
)
order by proname;
